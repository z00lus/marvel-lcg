## File Structure

Card scripts live under `cards/pack/<expansion>/<card_id>.py`:

```
cards/
  pack/
    __init__.py          # Shared imports for all card scripts
    core/
      __init__.py        # Just: from cards.pack import *
      spider_man/
        __init__.py      # Just: from cards.pack import *
        01001a.py        # Spider-Man Hero ability
        01001b.py        # Peter Parker Alter-Ego ability
        01002.py         # Black Cat Ally
        01003.py         # Backflip Event
        01005.py         # Swinging Web Kick Event
        01007.py         # Spider-Tracer Upgrade
        01008.py         # Web-Shooter Upgrade
        01009.py         # Webbed Up Upgrade
        01165.py         # Eviction Notice Obligation
```

**Naming convention:** Card IDs match the `card_id` field in `data/cards.json`. Hero face files end with `a` (hero) and `b` (alter-ego), e.g. `01001a.py` and `01001b.py`.

Every `__init__.py` in a pack folder should contain:

```python
from cards.pack import *
```

This gives your card script access to all game types, `AbilityFactory`, helpers, etc.

## Minimal Template

```python
from . import *

# Card Name

def GetAbilities() -> Sequence['Ability']:

    def my_ability(effect: 'Effect', message: 'Message.WhenPlayerInTurn') -> None:
        this = effect.this.CastTo(Event)
        # ... do something ...

    return [
        AbilityFactory.WhenInYourPlayTurn(
            AbilityType.HeroAction,
            my_ability
        )
    ]
```

Every card file must define `GetAbilities() -> Sequence['Ability']`. This function is called by the engine when the card enters play.

## Ability Types

The `AbilityType` enum controls when and how an ability triggers. Common ones:

| AbilityType | When it triggers | Example |
|-------------|-----------------|---------|
| `AbilityType.Action` | Player chooses to activate | "Choose → do X" |
| `AbilityType.HeroAction` | Hero form only, player activates | "Hero Action: Deal 3 damage" |
| `AbilityType.AlterEgoAction` | Alter-ego form only, player activates | "Alter-Ego Action: Draw 1 card" |
| `AbilityType.Interrupt` | Player chooses to interrupt a window | "Interrupt: Prevent 1 damage" |
| `AbilityType.ForcedInterrupt` | Must interrupt (no choice) | "Forced Interrupt: When this attacks..." |
| `AbilityType.HeroInterrupt` | Hero form interrupt | "Hero Interrupt: When you would take damage..." |
| `AbilityType.Response` | After an event happens | "Response: After you play an ally..." |
| `AbilityType.ForcedResponse` | Must respond (no choice) | "Forced Response: After this defeats a minion..." |
| `AbilityType.WhenRevealed` | When encounter card is revealed | "When Revealed: Deal 1 damage" |
| `AbilityType.WhenDefeated` | When this card is defeated | "When Defeated: Search your deck..." |
| `AbilityType.Resource` | Generates resources | "Discard → generate 1 [energy]" |
| `AbilityType.HeroResource` | Resource in hero form only | "Hero Resource: Exhaust → generate 1 [mental]" |
| `AbilityType.Rule` | Always-on constant ability | Passive effects, keywords |
| `AbilityType.Boost` | When this card is a boost card | "Boost: +1 ATK" |
| `AbilityType.Setup` | Game setup ability | "Setup: Search for..." |
| `AbilityType.NonKeyword` | Constant ability (not a keyword) | "You can only have 1 upgrade" |

## AbilityFactory Methods

`AbilityFactory` has static methods that create abilities for common trigger patterns.

### Player Turn Actions

```python
# Hero Action during your turn
AbilityFactory.WhenInYourPlayTurn(
    AbilityType.HeroAction,
    operation_fn,
    conditions=[]
)

# Alter-Ego Action
AbilityFactory.WhenInYourPlayTurn(
    AbilityType.AlterEgoAction,
    operation_fn,
)
```

### After Events

```python
# After a card enters play
AbilityFactory.AfterCardEnterPlay(
    AbilityType.Response,   # or ForcedResponse
    which_player,           # "You", "AnyPlayer", "Any"
    which_card,             # "This", Ally, Minion, etc.
    operation_fn,
)

# After a unit attacks
AbilityFactory.AfterUnitAttackAndDamageUnit(
    AbilityType.ForcedResponse,
    which_card,             # "This", Minion, etc.
    which_player,           # "You"
    operation_fn,
)

# After a unit defeats another unit
AbilityFactory.AfterUnitAttackAndDefeatUnit(
    AbilityType.Response,
    which_card,             # "This"
    which_enemy,            # Minion, Enemy
    operation_fn,
)

# After a unit defeats a scheme
AbilityFactory.AfterUnitDefeatedScheme(
    AbilityType.Response,
    which_card,             # "This"
    SchemeSide2,
    operation_fn,
    is_from_thwart=True
)

# After a player played a card
AbilityFactory.AfterPlayerPlayedCard(
    AbilityType.ForcedResponse,
    which_player,           # "You"
    which_card,             # "This"
    operation_fn,
)
```

### Interrupts (Preventing/Replacing)

```python
# When a unit would take damage
AbilityFactory.WhenUnitWouldTakeDamage(
    AbilityType.Interrupt,
    which_player,           # "You"
    operation_fn,
    is_from_attack=True
)

# When a unit would attack
AbilityFactory.WhenUnitWouldAttack(
    AbilityType.Interrupt,  # or ForcedInterrupt
    which_card,             # "This", "AttachedEnemy", Villain
    operation_fn,
)
```

### When Revealed / When Defeated

```python
# When this card is revealed
AbilityFactory.WhenThisRevealed(
    condition,              # None, "Alter-Ego", "Hero"
    operation_fn,
)

# When a card is revealed (any card matching filter)
AbilityFactory.WhenPlayerRevealCard(
    AbilityType.HeroInterrupt,
    which_player,           # "AnyPlayer"
    which_card,             # Treachery, Minion, etc.
    can_be_cancel,          # "All", "WhenRevealed", "Yes", None
    operation_fn,
    from_encounter=True,
)

# When this unit is defeated
AbilityFactory.WhenUnitBeDefeated(
    AbilityType.ForcedInterrupt,
    which_card,             # "This", "AttachedMinion"
    operation_fn,
)
```

### Upgrade / Support Attachment

```python
# Upgrade: attaches to a character
AbilityFactory.CanPlayThisUpgradeCard(
    target,                 # Minion, Enemy, Ally, None (defaults to "YourIdentity")
    conditions=[]
)

# Support: enters play under your control
AbilityFactory.CanPlayThisSupportCard(
    under_any_players_control=False,
    conditions=[]
)
```

### Resources

```python
# Generate resources (constant ability)
AbilityFactory.CanGenerateResources(
    AbilityType.Resource,   # or HeroResource, AlterEgoResource
    Resources("G"),         # G=Physical, Y=Energy, B=Mental, R=Wild
)
```

### Turn Phase

```python
# When your turn begins
AbilityFactory.WhenPlayerTurnBegin(
    AbilityType.ForcedResponse,
    which_player,           # "You", "Any"
    operation_fn,
)

# When your turn ends
AbilityFactory.WhenPlayerTurnEnd(
    AbilityType.ForcedResponse,
    which_player,
    operation_fn,
)
```

## The Effect & Message Pattern

Every ability handler receives `(effect, message)`:

```python
def my_ability(effect: 'Effect', message: 'Message.WhenPlayerInTurn') -> None:
    # effect.this     → the card face that owns this ability
    # effect.targets  → the selected targets (list of CardFace)
    # effect.initiator → who initiated the action (Player or CardFace)
    # effect.world    → the World instance

    this = effect.this.CastTo(Ally)  # Cast to a specific card type

    # Do something with targets
    this.DealDamage(effect.targets, 3, effect)
```

### Key effect properties

| Property | Type | Description |
|----------|------|-------------|
| `effect.this` | `CardFace` | The card face that owns this ability |
| `effect.targets` | `List[CardFace]` | Selected targets (from `.SetTarget()`) |
| `effect.initiator` | `Player\|CardFace` | Who initiated the action |
| `effect.cost_func` | `CostFunc` | Access to cost function results |
| `effect.world` | `World` | The game world |
| `effect.GetInitiator()` | `Player` | Get the player who initiated |

### Key message properties

| Message | Key properties |
|---------|---------------|
| `Message.WhenPlayerInTurn` | `message.to_player` |
| `Message.AfterCardEnterPlay` | `message.trigger` (the card that entered) |
| `Message.WhenUnitWouldTakeDamage` | `message.PreventDamage(...)` |
| `Message.WhenPlayerRevealCard` | `message.trigger`, `message.GetToPlayer()`, `message.CancelWhenRevealedEffect(...)` |
| `Message.WhenCardRevealed` | `message.trigger` (the revealed card) |
| `Message.AfterUnitAttackUnit` | `message.GetToPlayer()` |

### The `this` pattern

Always start with `effect.this.CastTo(CardType)` to get a typed reference. Use `Unused(this)` if you don't use it directly (avoids lint warnings):

```python
def my_ability(effect: 'Effect', message) -> None:
    this = effect.this.CastTo(Ally)
    Unused(this)
    # ... ability logic ...
```

## Fluent API

Abilities use method chaining for configuration:

```python
AbilityFactory.WhenInYourPlayTurn(
    AbilityType.HeroAction,
    my_operation
)
.SetName("Ability Name")           # Display name
.SetPlay()                        # Requires playing the card from hand
.SetLabel('attack')               # Label as attack/thwart/defense
.SetCost(Cost("Y"))               # Resource cost
.SetCostFunc(CostFunc.Exhaust("This"))  # Exhaust this card as cost
.SetTarget(Enemy)                 # Set targeting
.SetTarget2("YourIdentity")       # Target yourself
.LimitOncePerRound()              # Can only use once per round
```

### Common chain methods

| Method | Description |
|--------|-------------|
| `.SetName(str)` | Set the ability's display name |
| `.SetPlay()` | Card must be played from hand to use |
| `.SetPlay(hero_form_only=True)` | Only usable in hero form |
| `.SetLabel(str)` | Label as `"attack"`, `"thwart"`, or `"defense"` |
| `.SetCost(Cost(...))` | Set resource cost |
| `.SetCostFunc(CostFunc.*)` | Add a cost function |
| `.SetTarget(...)` | Set target selector |
| `.LimitOncePerRound()` | Once per round limit |
| `.LimitMaxPerTurn(int)` | Max N times per turn |

## Cost Functions

Cost functions define what the player must pay to use an ability:

```python
# Exhaust this card
CostFunc.Exhaust("This")

# Exhaust your identity
CostFunc.Exhaust("YourIdentity")

# Discard cards from hand
CostFunc.Discard("YourHandCards")

# Remove a counter
CostFunc.Counter("This", 1, "web")    # Remove 1 "web" counter

# Heal a character
CostFunc.Heal(1, "This")
```

Multiple cost functions can be chained:

```python
AbilityFactory.CanGenerateResources(
    AbilityType.HeroResource,
    Resources("G")
).SetCostFunc(CostFunc.Exhaust("This"))
 .SetCostFunc(CostFunc.Counter("This", 1, "web"))
```

## Targets & Selectors

Rules Reference 1.8 allows a target for a multi-effect ability when at least
one part of that ability can affect it. Use `affects_target_if` when several
independent instructions affect the same selected target:

```python
.SetTarget(
    Enemy,
    affects_target_if=(
        Condition.TargetCanBeConfused,
        Condition.TargetCanTakeDamage,
    ),
)
```

The functions in `affects_target_if` are alternatives (OR). The normal finder,
attack/thwart legality, Crisis, ownership, and control checks are still all
required. Do not use this option to implement a **then** instruction: the code
resolving a then instruction must still check that its preceding instruction
succeeded before resolving the dependent effect.

### Target strings

| String | Meaning |
|--------|---------|
| `"This"` | This card |
| `"You"` | The current player |
| `"AnyPlayer"` | Any player can be targeted |
| `"YourIdentity"` | Your identity card |
| `"Players"` | All players |
| `"Initiator"` | The initiating player |
| `"Trigger"` | The card that triggered this ability |
| `"AttachedEnemy"` | The enemy this upgrade is attached to |
| `"AttachedMinion"` | The minion this upgrade is attached to |
| `"YourHandCards"` | Cards in your hand |

### Card type targets

```python
Enemy          # Any enemy (villain or minion)
Ally           # Any ally
Minion         # Any minion
Scheme2        # Any side scheme
MainScheme     # The main scheme
Upgrade        # Any upgrade
Support        # Any support
```

### Range modifiers

```python
.SetTarget(Enemy, range="All")           # All enemies
.SetTarget(Scheme2, range=("Zero", 3))   # 0-3 schemes
.SetTarget(Enemy, canbe_confused=True)   # Can apply confused
.SetTarget("This", canbe_ready=True)     # Can ready
```

## Buffs

For tracking custom state across rounds, use the `Buff` class:

```python
class MyBuff(Buff):
    @override
    def __init__(self) -> None:
        super().__init__()
        self.some_state = False

    @override
    def OnRecordPlayedFace(self, face: 'CardFace'):
        super().OnRecordPlayedFace(face)
        # Track when a card is played
        if Ally.IsType(face):
            self.some_state = True
            self.SetUIText(face.paper.card_id)

    @override
    def OnRoundEnd(self):
        super().OnRoundEnd()
        self.some_state = False
        self.SetUIText("")

    def __bool__(self) -> bool:
        return not self.some_state
```

### Obligation — Eviction Notice

from . import *

# Eviction Notice

def GetAbilities() -> Sequence['Ability']:

    def eviction_notice(effect: 'Effect', message: 'Message.WhenObligationGiveToPlayer') -> None:
        this = effect.this.CastTo(Obligation)
        Unused(this)
        player = message.GetGaveToPlayer()

        YouMayFlipToYourAlterEgoForm(player, effect)

        def discard_random_handcards_and_surge(targets: Sequence['CardFace']) -> None:
            player.DiscardRandomHandCards(1, effect)
            this.GainSurge(1, effect)
            Faces.DiscardAll([this], effect)

        player.ChooseAbilities(

            AbilityFactory.ForChoiceAbility(
                "Exhaust Peter Parker → remove Eviction Notice from the game",
                lambda targets: Faces.RemoveAllFromGame([this], effect),
                condition=player.IsAlterEgo(),
            ).SetCostFunc(CostFunc.Exhaust("YourIdentity")),
            AbilityFactory.ForChoiceAbility(
                "Discard 1 card at random from your hand. This card gains surge. Discard this obligation",
                discard_random_handcards_and_surge
            )
        )

    return [
        AbilityFactory.WhenThisObligationGiveToYou(
            eviction_notice
        )
    ]

### Complex Card with Buff — Steve Rogers

```python
from . import *

# * Steve Rogers

def GetAbilities() -> Sequence['Ability']:

    class Buff_03001b(Buff):
        @override
        def __init__(self) -> None:
            super().__init__()
            self.played_ally = False

        @override
        def OnRecordPlayedFace(self, face: 'CardFace'):
            super().OnRecordPlayedFace(face)
            if Ally.IsType(face):
                self.played_ally = True
                self.SetUIText(face.paper.card_id)

        @override
        def OnRoundEnd(self):
            super().OnRoundEnd()
            self.played_ally = False
            self.SetUIText("")

        def __bool__(self) -> bool:
            return not self.played_ally

    def steve_rogers_setup(effect: 'Effect', message: 'Message.WhenCardSetup') -> None:
        initiator = effect.GetInitiator()
        face = Search.PlayerCard(
            effect, initiator,
            include_player_deck=True,
            include_discard_pile=True,
            name=CAPTAIN_AMERICAS_SHIELD,
            card_type=Upgrade,
        )
        if face:
            initiator.GainCard(face, effect)

    return [
        AbilityFactory.WhenCardSetup("This", steve_rogers_setup),
        AbilityFactory.ReduceCostToPlayFaceWhen(
            Ally, 1, "AnyPlayer",
            conditions=[
                lambda effect, message:
                    bool(effect.this.GetBuff(Buff_03001b))
            ],
        ).SetName("Living Legend"),
    ]
```

## Adding a New Expansion

1. Create a folder under `cards/pack/` with your expansion abbreviation:
   ```
   cards/pack/my_exp/
   ```

2. Create `__init__.py`:
   ```python
   from cards.pack import *
   ```

3. Create card files (one per card face):
   ```
   cards/pack/my_exp/01001a.py  # Hero face
   cards/pack/my_exp/01001b.py  # Alter-Ego face
   cards/pack/my_exp/01002.py   # Ally
   cards/pack/my_exp/01003.py   # Event
   ```

4. Register cards in `data/cards.json` with matching `card_id`, `type`, `name`, etc.

5. Add card images to `assets/pics/` with matching IDs.
